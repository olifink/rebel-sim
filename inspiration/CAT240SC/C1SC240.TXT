code acompare  ( a b == | -- flag{-1,0,1} <<or>> a b == NaN )

   begin,  a1 ) d1 move,  a0 ) a1 )+ move,  d1 a0 )+ move,  d0 nt -until,

   then,  1 #n d1 subq,  eq

code pack  ( addr length integerlen sign -- flag )

   if,  d0 d3 move,  1 #n d3 addq,  then,
   awhole 1+ #n d3 cmp,  pl   ( too many integers? )
   if,  -1 #n sp ) move,  0c #n a1 ) .b or,  next, ( make uNaN, exit )
   then,  d0 d2 move,  d3 d2 sub,
   awide 2* awhole - 2- #n d2 cmp,  pl

   awhole 1+ #n d1 moveq,  d3 d1 sub,

   1 #n d2 lsr,  d2 a1 add,   ( a1 = dest byte )

   a0 )+ d2 .b move,  i' dpoint 3 + d2 .b cmp,  ne until,
   i' commapun 3 + d2 .b cmp,  ne until,

   1 #n d1 and,  eq

   d3 d2 .b or,  d2 a1 ) .b move,
   else,  0f0 #n d3 .b and,  d3 d2 .b or, ( > low  nybble )

: chk_timer 
( name -> f | check timer <name>, if elapsed returns false,if not elapsed true )
   dup 0=
   if    ( timer is 0, not expired )
      drop true exit
   else

      if    ( timer expired )
         false exit
      else  ( timer not expired )
         true exit
      then
   then

   ticks/sec * ticks@ + swap !

: recch?
( | returns true if a character can be read via recget, else returns false)

   rxch?

: recget
( -> c | returns an LT data character received by MNP on each call, clears 
data.up when last character delivered.  If MNP not running, passes data from 
modem)

connect.ind
if       ( MNP is running)
   data.up
   if
      outptr c@ 1 outptr +to
      -1 out.count +to
      outbuf.full off      ( outbuff no longer full)
      out.count 0=
      if                ( last char delivered, clear data.up )
         data.up off
      then
   else
      7 " recget/fatal error data.up not set" dbsc
      noop              ( what should we do here ? )
   then
else     ( MNP not running, pass modem info thru)
   rxget
then
;

( 4.2 Frame Disassembly Word 
#####################################################################


frame_disassembly is the first word called upon detection of an end-of-frame.  
The frame is disassembled, FCS calculated, and the LPDU extracted.

Susidiary words used in frame_disasssembly -- )

 : dle_parsing
(  | if the current char is DLE, do transparency, end flag checking )
dv1 " fr_dis/dle_p" dbs space
   curchar DLE =
   if                   ( 1st char is DLE )
      rxchar no_ch_ex dup DLE =
      if                   ( there are two DLEs )
         cal_crc           ( CRC calculation for the DLE )
         drop
      else                 ( 1st char is DLE, 2nd is not )
         dup ETX =
         if                ( ETX, end of frame )
            ( we have end of frame )
            frame.stat 1 > if exit then ( frame no good, return )
            curchar to
            cal_crc        ( CRC calculation for the ETX )
            rxchar no_ch_ex rxchar no_ch_ex     ( get FCS chars )
            swap 8 shl +   ( form 16 bit FCS value )
            dup dv1 " dle_parsing/FCS val =" dbk space
            curval dv2 " CRC val=" dbkc
            curval - 0=
            if             ( calc CRC = FCS )
               dv1 " dle_parsing/end of good frame" dbsc
               frame.stat on  ( we have a good frame )
               exit
            else           ( bad CRC )
               ngf.crc frame.stat to
               dv1 " dle_parsing/end of frame bad CRC" dbsc
               ( DB  frame.stat on  # force bad CRC frame to look OK )
               exit
            then
         else              ( not ETX, single DLE )
            dv3 " dle_parsing/end of frame single DLE error" dbsc
            one.dle frame.stat to   ( single DLE, an error )
            exit
         then
      then
   then
;

: send_it
( | There is a frame in the transmit buffer, ship it )
    dv2 " send_it ->" dbsc
   local count
   xmt.count count to
   xmitbuff xmtptr to
   begin
      xmtptr c@ txchr 1 xmtptr +to -1 count +to
      ( count dv2 "  " dbkc )
      0 count =
      if leave then
   again
   cr
;

   frame.stat dv2 " send_ld/frame.stat=" dbkc
   build_ld_skel
   send_it
   to_idle
;

 dv2 " debug version of send shutup LA word" dbsc
   local count

: frame_disassembly

   dv2 " fr_dis/start frame_disassembly " dbsc
   rcvbuff rcvptr to ( reset receive buffer pointer )
   2 rcvptr +to      ( skip over 1st two receive buffer cells,
                      save for char count )
   0 sflg.retry.ctr to           ( initialize retry counter )
   fr.timer addr assem.delay set_timer ( initialize frame timer )
   0 frame.stat to         ( set to 0, frame not finished )
   begin    ( start disassembly loop )
      dv1 " fr_dis/start disassem loop" dbsc
      sflg.retry.ctr retry.lim >
      if          ( too many retries )
         ngf.retries frame.stat to
         exit
      then
      begin          ( loop - find frame start flag )
         dv2 " fr_dis/start find frame loop" dbsc

         false gotframe to    ( clear the got-a-frame flag )
         rxchar no_ch_ex SYN =
         if                ( we have SYN )
            rxchar no_ch_ex DLE <>
            if             ( not DLE )
               1 sflg.retry.ctr +to
               leave       ( not DLE, try again )
            else           ( we have DLE ) 
               rxchar no_ch_ex STX <>
               if          ( not STX )
                  1 sflg.retry.ctr +to
                  leave       ( not STX, restart )
               else
                  ( header flag, a frame, start char-by-char processing )
                  curstrt curval to ( initialize CRC current value )
                  true gotframe to  ( set the got-a-frame flag )
                  dv2 " fr_dis/got a frame, start processing" dbsc 
                  begin
                     rxchar no_ch_ex curchar to       ( get next input char )
                     ( DB  curchar . space )
                     dle_parsing
                     no_ch_ex
                     frame.stat 1 > if exit then      ( frame no good )
                     frame.stat true = if leave then  ( frame good + finished )
                     cal_crc                 ( CRC calculation for this char )
                     curchar rcvptr c! ( save this char, bump pointer )
                     1 rcvptr +to   
                     rcvptr rcvbuff - frame.limit >
                     if ngf.flim frame.stat to exit then ( frame too big )
                  again ( loop end - char-by-char frame processing )
               then
            then
         else
            ( DB ) fr.timer chk_timer . space
            dv2 " fr_dis/no SYN, retry" dbsc

            1 sflg.retry.ctr +to    ( 1st char not SYN, try again )
            leave
         then
         true gotframe = if leave then ( exit loop if we found a frame )
         dv2 " fr_dis/no frame, retry"  dbsc
      again             ( loop end - start flag find )
      fr.timer chk_timer
      if                ( timer not expired )
         true gotframe =
         if             ( we've found a frame )
            leave
         then
      else              ( timer expired )
         dv2 " fr_dis/timer expired" dbsc
         ngf.time frame.stat to
         exit
      then
   again    ( loop end - disassembly loop )
   ( we have received a frame )
    dv2 " fr_dis/we have a frame"  dbsc

   ( rcvptr is on the last received byte; calculate size of frame,
    store it as two successive bytes, LSB first )
   rcvptr rcvbuff -
   2 -                     ( allow for size bytes )
   dup dv2 " fr_dis/fr size=" dbkc
   ( NB frame size is 1 larger than LI, LI doesn't include LI byte )
   dup
   lo.mask and             ( mask the lower size byte )
   rcvbuff rcvptr to       ( reset the pointer )
   rcvptr c! 1 rcvptr +to  ( store the lower size byte, bump ptr )
   8 shr                   ( shift upper size byte right )
   hi.mask and             ( mask upper size byte )
   rcvptr c!               ( store it )

   rcvbuff rcvptr to ( reset receive buffer pointer )
   2 rcvptr +to      ( skip over char count cells )
   ( get PDU parms from frame )
   rcvptr c@ hdrlen to 1 rcvptr +to ( save LI, the header length )
   rcvptr c@ pdutyp to 1 rcvptr +to ( save the PDU functional type )


hdrlen dv2 " fr_dis/hdrlen=" dbk space
pdutyp dv2 " fr_dis/pdutyp=" dbk space

state dv2 " fr_dis/state =" dbk space
dv2 " end of frame_disassembly" dbsc
dv2 dbug.lev >
if
   rcvb_read
then
;

( 4.3 Link Establishment Word 
#####################################################################


link_establishment sets up the link by carrying out the 3-way handshake, 
performing LR parm negotiation, etc.  NB - handle bad frame - frame.stat <> 
true. Reset frame.stat to 0 when PDU delivered to buffer, or exit on fail.

Susidiary words used in link_establishment -- )

: move_ptrs
( n -> | move rcv, xmit ptrs from start of buffers to offset n )
   dup            ( dup offset for xmit use )
   rcvbuff rcvptr to    ( initialize the pointers )
   xmitbuff xmtptr to
   rcvptr +to     ( move rcv ptr to the offset point )
   2 -            ( allow for 2 extra cells in rcv buffer )
   xmtptr +to     ( move xmit ptr to the offset point )
;

: bad_parm
( f n -> | if parm is bad, send LD with bad parm reason n, exit )
dup dv2 " bad_parm" dbkc
swap if send_ld exit then
;

: data_state
( | do what we need for data transfers )
   st.data state to           ( we're ready for transfers, go to data state )
   data.in on data.out off    ( go to input state )
   connect.ind on             ( tell send/receive we've accepted a link )
;

: link_establishment
dv2 " lkes/start link_establishment " dbsc

   true connect.req = 
   if                ( we have a connection request from send/receive )
      dv2 " lkes/we have connect.req" dbsc
      connect.req off   ( clear flag so send/receive knows we have responded )
      state st.data =
      if             ( we are already in data state )
         ( NB this should never occur, but if it does, )
         reset_all
         exit
      then
      data.out on data.in off ( go to output state [do we need both flags ?] )
      build_lr_skel ( build an LR )
      send_it
      st.lr.resp.wait state to
      lr.init.timer addr lr.delay set_timer
      exit
   then  ( end if - connect.req test )

   ( use PDU type to see what to do next )
   state st.idle =
   if                   ( we are in idle )
      dv2 " lkes/in idle" dbsc
      lr.pdu pdutyp =
      if                ( PDU is an LR )
         true frame.stat to      ( initialize frame.stat to good frame value )

         ( We are in idle and have an LR. We are accepting a link request.
          Check the parms in the received LR.  Send our skeleton LR if
          the parms in the received LR are OK. )

         st.parms.nego state to  ( set negotiate state )
         ( negotiate protocol level )
         ( we always want 2, standard level, it is already in the LR
          skeleton, continue )
         ( negotiate serial number )
         ( any serial number is valid, continue )
         svcl.off move_ptrs               ( ptrs to service class value )
         rcvptr 2 - rcvptr to          ( move rcv ptr to svc class parm code )
         rcvptr c@ lr.svclas.code <>   ( is parm code OK )
         if                            ( svc class bad )
            ngf.code frame.stat to
            weird.parm send_ld exit       ( not service class code, exit )
         then
         1 rcvptr +to rcvptr c@ len.1 <>  ( is parm length OK )
         if                               ( length bad )
            ngf.len frame.stat to
            weird.parm send_ld exit
         then
         ( negotiate service class )
         svcl.off move_ptrs            ( move recv pointer to svc class value )
         ( rcvptr c@ dv2 " lkes/svc class =" dbkc )
         rcvptr c@ lr.svc.clas <>
         if                            ( svc class bad )
            ngf.svc.cla frame.stat to
            weird.parm send_ld exit
         then
         ( dv2 " lkes/post svc cl nego" dbsc )
         ( negotiate credit allocation )
         cred.off move_ptrs               ( ptrs to credit allocation value )
         rcvptr 2 - rcvptr to          ( rcv ptr to cred alloc parm code )
         rcvptr c@ lr.cred.code <>     ( is parm code OK for credit alloc )
         if                            ( code bad )
            ngf.code frame.stat to
            weird.parm send_ld exit
         then
         1 rcvptr +to rcvptr c@ len.1 <>  ( is parm length OK )
         if                               ( length bad )
            ngf.len frame.stat to
            weird.parm send_ld exit
         then
         ( we always want 1 for credit allocation, it is already in the
          LR skeleton, continue )
         ( negotiate max data size )
         maxd.off move_ptrs               ( ptrs to max user data size value )
         -2 rcvptr +to                 ( rcv ptr to max user data parm code )
         ( DB  ." lkes/maxd code=" rcvptr c@ . space )
         rcvptr c@ lr.maxdat.code <>   ( is parm code OK for max data )
         if                            ( code bad )
            ngf.code frame.stat to
            weird.parm send_ld exit
         then
         1 rcvptr +to rcvptr c@ len.2 <>  ( is parm length OK )
         if                               ( length bad )
            ngf.len frame.stat to
            weird.parm send_ld exit
         then
         ( we always want 40 for max size, it is already in the
          LR skeleton, continue )
         ( negotiate data phase optimization )
         daphop.off move_ptrs          ( ptrs to data phase opt. value )
         -2 rcvptr +to                 ( rcv ptr to data phase opt. parm code )
         rcvptr c@ lr.daphop.code <>   ( is parm code OK for data phase opt. )
         if                            ( code bad )
            ngf.code frame.stat to
            weird.parm send_ld exit
         then
         1 rcvptr +to rcvptr c@ len.1 <>  ( is parm length OK )
         if                               ( length bad )
            ngf.len frame.stat to
            weird.parm send_ld exit
         then
         ( we always want 3 for data phase value, it is already in the
          LR skeleton, continue )
      else     ( PDU is not an LR )
         ld.rsn.nlr send_ld            ( 3-way proto error )
         exit
      then  ( end if - is it LR )
      frame.stat
      if       ( negotiation was OK )
         dv2 " lkes/acc/send res LR" dbsc
         build_lr_skel
         send_it           ( transmit buffer contains the response LR, ship it )
         endflag.req off   ( we're ready for the next incoming frame )
         st.la.wait state to  ( go to LA_WAIT state )
      else     ( negotiation failed )
         exit
      then
   else  ( we are not in idle )
      ( DB # ." lkes/not in idle" cr )
      st.lr.resp.wait state =
      if    ( we are in LR_RESP_WAIT state waiting for response to our LR )
         state dv2 " lkes/state is resp.wait, state =" dbkc
         lr.pdu pdutyp =
         if                ( we have an LR )
            dv2 " lkes/we have the resp LR" dbsc
            ( We have received an LR in response to the LR we sent,
             negotiate the parms.  This means in most cases getting
             what we want or disconnecting. )
            ( NB should we check parm codes, lengths ? the code below
             just looks at the parm values )
            st.parms.nego state to
            ( negotiate protocol level. We always want 2, standard )
            rcvbuff prol.off + rcvptr to  ( rcv ptr to protocol level )
            ( if protocol level not 2, set ldreason and quit )
            rcvptr c@ lr.prot.lvl <>
            if
               ngf.prot frame.stat to
               ld.rsn.inc send_ld exit
            then
            ( negotiate service class )
            rcvbuff svcl.off + rcvptr to  ( rcv ptr to service class )
            ( discon + quit if svc class 0 or > 2 )
            rcvptr c@ dup lr.svc.clas >
            if
               ngf.svc.cla frame.stat to
               ld.rsn.unk send_ld exit
            then
            1 <
            if
               ngf.svc.cla frame.stat to
               ld.rsn.unk send_ld exit
            then
            ( negotiate credit allocation )
            rcvbuff cred.off + rcvptr to  ( rcv ptr to credit allocation )
            rcvptr c@ lr.crd.allo <>
            if
               ngf.cred frame.stat to
               ld.rsn.unk send_ld exit
            then
            ( negotiate max user data size )
            rcvbuff maxd.off + rcvptr to  ( rcv ptr to max user data 
                                            size, lo byte )
            rcvptr c@ lr.mdata.lo <>
            if
               ngf.mdata frame.stat to
               ld.rsn.unk send_ld exit
            then
            1 rcvptr +to                  ( advance pointer to hi byte )
            rcvptr c@ lr.mdata.hi <>
            if
               ngf.mdata frame.stat to
               ld.rsn.unk send_ld exit
            then
            ( negotiate data phase optimization )
            rcvbuff daphop.off + rcvptr to   ( rcv ptr to data phase opt. )
            rcvptr c@ lr.data.ph <>
            if
               ngf.dap frame.stat to
               ld.rsn.unk send_ld exit
            then
               ( the LR parms all check OK, send the LA )
            1 local.cred to   ( set the initial credit to 1 )
            0 local.seqn to   ( set the initial sequence number to 0 )
            dv2 " lkes/pre LA send" dbsc
            build_laf
            send_it
            endflag.req off   ( we're ready for the next incoming frame )
            st.data state to  ( we're ready for transfers, go to data state )
            data.out on data.in off    ( go to output state )
            state dv2 " lkes/leaving link est., state=" dbkc
            1 remote.cred to     ( give ourselves credit for 1st LT )
            start.lt on          ( prepare to build 1st LT )
            exit
         else        ( not an LR, a failure )
            ngf.nlr frame.stat to
            ld.rsn.nlr send_ld exit
         then  ( end if - LR test )

         lr.init.timer chk_timer 0=
         if       ( timer is expired )
            dv2 " lkes/timer expired." dbsc
            lr.retry.ctr max.lr.ret <
            if    ( not max retries )
               dv2 " lkes/not max retries, send LR again." dbsc
               1 lr.retry.ctr +to
               send_it                                ( send the LR again )
               lr.init.timer addr lr.delay set_timer  ( initiate the LR timer )
            else        ( sent LR max number of times, discon + quit )
               ld.rsn.rex send_ld         ( retransmission limit proto error )
            then
            else     ( timer hasn't expired )
            ( DB # ." lkes/timer not expired" cr )
         then  ( end if - timer expired test )
      then  ( end if - LR RESP_WAIT state test )
      st.la.wait state =
      if    ( we are in LA_WAIT state )
         pdutyp la.pdu =
         if             ( we have an LA )
            dv2 " lkes/in LA_WAIT, an LA" dbsc
            rcvbuff l.seqn.off + rcvptr to
            rcvptr c@ remote.seqn to      ( get seq num from LA )
            rcvbuff la.cred.off + rcvptr to
            rcvptr c@ remote.cred to      ( get credit from LA )
            endflag.req off         ( we're ready for the next incoming frame )
            data_state
         else
            pdutyp lt.pdu =
            if          ( not an LA, it is an LT )
               dv2 " lkes/in LA_WAIT, an LT" dbsc
               data.in on data.out off    ( go to input state )
               ( NB the code below bumps seq num and fakes cred of 1
                so that this first LT will be correctly handled by
                data_transmission )
               rcvbuff l.seqn.off + rcvptr to
               rcvptr c@ remote.seqn to      ( get seq num from LT )
               ( rcvptr c@ 1 + remote.seqn to      # get seq num from LT )
               remote.seqn dv2 " data_t/remote.seqn=" dbkc
               1 remote.cred to
               endflag.req off      ( we're ready for the next incoming frame )
            else        ( not an LA or an LT, disconnect )
               ngf.nlat frame.stat to
               ld.rsn.nlr send_ld exit
            then
         then  ( end if - we have an LA )
      then  ( end if - in LA_WAIT state )
   then  ( end if - idle test )
;

( 4.4 Data Transmission Word 
#####################################################################


data_transmission handles sending and/or receiving data after a link has been

( Susidiary words used in data_transmission -- )

: count_gen
( | generates the char count from the receive buffer, leaves ptr on byte after 
count )
   rcvbuff rcvptr to
   ( rcvptr c@ dv2 " count_gen/lo ord byte=" dbkc )
   rcvptr c@ 1 rcvptr +to     ( get lo order byte )
   ( rcvptr c@ dv2 " count_gen/hi ord byte=" dbkc )
   rcvptr c@ 1 rcvptr +to     ( get hi order byte )
   8 shl + rcv.count to    ( combine bytes )
;

: ltx_adv
( c -> | load next LT buffer cell )
   ltxptr c! 1 ltxptr +to
;

: ldlt_char
( c -> | put the char in the next LT buffer cell, doing CRC )
curchar to cal_crc curchar ltx_adv
;

: send_lt
( | There is a frame in the LT buffer, ship it )
   dv2 " send_lt ->" dbsc
   local max
   local count
   xmitbufsize ltxbuff + max to
   xmitbufsize count to
   ltxbuff ltxptr to
   begin
      ( test for end of LT )
      ltxptr c@ dup DLE =
      if                         ( char is DLE )
         txchr 1 ltxptr +to      ( send DLE )
         ltxptr c@ dup ETX =
         if                      ( char after DLE is ETX )
            txchr 1 ltxptr +to   ( send ETX )
            2 count to           ( prepare to send FCS chars )
         then
      else
         txchr 1 ltxptr +to      ( an ordinary char, send it )
      then
      3 count >
      if                         ( we got a DLE ETX, send the FCS chars )
         -1 count +to
         0 count >
         if                      ( FCSs sent, exit )
            exit
         then
      then

      ltxptr max >
      if       ( never found the ETX - just keep on truckin )
         dv2 " send_lt/no ETX in this LT " dbsc
         exit
      then
   again
;

: build_lt_skel
( | build the start flag for a fixed field LT )
   ltxbuff ltxptr to
   SYN ltx_adv          ( load the start flag, not included in CRC )
   DLE ltx_adv
   STX ltx_adv
   2 ldlt_char          ( load LI.  There are 2 octets in the LT header )
   lt.pdu ldlt_char     ( load LT type code )
   local.seqn ldlt_char ( load sequence number )
   ( pointer now set to 1st data cell )
;

: ship_lt
( | append stop flag, calculate FCS, send the LT in ltxbuff )
   DLE ltx_adv    ( load the stop flag )
   ETX ldlt_char
   curval 8 shr   ( shift the MS byte of FCS into the LS byte )
   ltx_adv
   curval ltx_adv ( put the LS byte in the buffer )
   send_lt
   lt.timer addr lt.delay set_timer    ( start the LT timer )
   0 remote.cred to  ( sent an LT, used up our credit, set it to 0 )
   no.xmit.ind on    ( tell send/receive we can't send )
   ltxbuff ltxptr to ( reset the pointer )
;

: data_transmission

( | data_transmission handles sending and/or receiving data after a link has 
been established. )

dv2 " data_t/start data_transmission " dbsc

carrier? 0=
if          ( no carrier from the modem)
   dv3 " data_t/no modem carrier " dbsc
   to_idle
   exit
then

0 trans.stat to      ( initialize error value )

data.in
if ( We are receiving data. We receive LTs and send LAs.  We receive LNs and 
send LNAs. )

   la.timer dv2 " data_t/la.timer=" dbkc
   la.timer chk_timer
   if                      ( lost LA timer has not expired )
      shutup.req
      if                   ( we have a shutup request )
         0 local.cred to   ( set credit to 0 )
         build_las         ( build the shutup LA, put it in its buffer )
         send_shut         ( NB may not be any if IRQ does this )
      then
      pdutyp ln.pdu =
      if                ( we have an LN )
         dv2 " data_t/received an LN" dbsc
         break.ind on   ( LNs always signal BREAK )

            send_it

            la.timer addr la.delay set_timer    ( initiate the lost LA timer )
            ngt.seqn trans.stat to  
            exit

               dv2 " data_t/post ACK LA" dbsc
               0 la.retry.ctr to
            then        ( end if - data.up test )
         then     ( end if - sequence number mismatch )

         ngt.not.lt trans.stat to

                  leave
               then
            else  ( no char to send )
               leave
            then
         again
         dv2 " post ship LT loop" dbsc
         exit
      else        ( no credit )
         exit
      then
   else        ( LT retransmit timer expired )
      lt.retry.ctr max.retries <
      if    ( not max retries )
         dv2 " data_t/not max retries, send LT again." dbsc
         1 lt.retry.ctr +to
         send_lt                             ( send the LN again )
         lt.timer addr lt.delay set_timer    ( initiate the LR timer )
      else        ( sent LT max number of times, discon + exit )
         ld.rsn.rex send_ld            ( retransmission limit proto error )
         dv2 " data_t/max LT retries, exit." dbsc
         exit
      then
   then

   pdutyp lna.pdu =
   if                ( we have an LNA )
      ( NB any chance of a rogue LNA not preceded by an LN ? )
      dv2 " data_t/received an LNA" dbsc
      rcvbuff rcvptr to
      ln.seqn.off rcvptr +to     ( get the LNA sequence number )
      rcvptr c@ loc.ln.seqn <>
      if                         ( seq number mismatch )
         dv2 " data_t/LN seq no mismatch" dbsc
         ngt.ln.seq trans.stat to
         exit
      then
      ( This is the LNA in response to the BREAK LN we sent. Maintain the
       link but forget any pending activity. Clear buffers, cancel timers )
      clear_buffers
      0 ln.timer to  ( cancel the LN timer )
      break.req off
      clean_up
   then
then  ( end if - sending data )
