code set.dtr   ( -- )
 sr sp -) .w move,               ( disable interrupts first )
 2700 #n sr .w move,
 i' system.status a1 .l move,    ( Get the paramater base address )
 a1 isb )d .l d6 move,           ( get isb )
 duart #n a0 .l move,            ( set up duart address )
 02 #n a0 ser.sopb )d .b move,   ( enable dtr )
 ser.dtr #n d6 .l bset,          ( and set in isb )
 d6 a1 isb )d .l move,           ( restore isb )
 sp )+ sr .w move,
 next;

code clr.dtr   ( -- )
 sr sp -) .w move,               ( disable interrupts first )
 2700 #n sr .w move,
 i' system.status a1 .l move,    ( Get the paramater base address )
 a1 isb )d .l d6 move,           ( get isb )
 duart #n a0 .l move,            ( set up duart address )
 02 #n a0 ser.ropb )d .b move,   ( disable dtr )
 ser.dtr #n d6 .l bclr,          ( and clear in isb )
 d6 a1 isb )d .l move,           ( restore isb )
 sp )+ sr .w move,
 next;

code tst.dtr   ( -- flag )
 i' system.status a1 .l move,    ( Get the parameter base address )
 a1 isb )d .l d6 move,           ( get isb )
 ser.dtr #n d6 .l btst,          ( see if dtr is set )
 eq if, 0 #n d0 moveq, else, -1 #n d0 moveq, then,
 d0 sp -) move,                  ( push flag )
 next;

 duart #n a0 .l move,            ( get base of duart )
 a0 ser.ip )d d0 .b move,        ( get flag register )
 20 #n d0 .b and,                ( see if dcd is set, bit 5 )
 eq if, -1 #n d0 moveq, else, 0 #n d0 moveq, then,
 d0 sp -) move,                  ( push flag )
 next;

code ser.?2chrs      ( -- size, true if two chars 0<char<b0 in buffer )
   i' system.status a1 move,            ( Get base address of the system area )
   a1 ser.rxchar )d d0 .w move,         ( Get pointer to character )
   d0 d3 .w move,                       ( save output pointer in d3 )
   a1 ser.rxptr )d d2 .w move,          ( get input pointer )
   a1 ser.rxbuf )d a0 move,             ( Get buffer address )
   0 #n d1 moveq,                       ( Clear the upper bits )
   1 #n d0 .w addq,                     ( pre-advance pointer )
   a1 ser.rxbuf.len )d d0 .w cmp, eq    ( end of buffer? )
   if, 0 #n d0 moveq, then,             ( clear the pointer )
   begin,
      d0 d2 .w cmp,                             ( see if hit input pointer )
      eq if, 0 #n sp -) move, next, then,       ( if so, not 2 chars, done )
      a0 d0 0 xw)d d1 .b move,                  ( Get the character )
      &lastasc #n d1 .w cmp,                    ( see if legal char )
      le if, d3 d0 .w sub,                      ( find distance between in&out)
            mi if, a1 ser.rxbuf.len )d d0 .w add, then, ( if wrap around )
            d0 swap, 0 #n d0 .w move, d0 swap,  ( clear out high bits )
            d0 sp -) move, next, then,          ( if so, have 2 chars, done )
      1 #n d0 .w addq,                          ( increment the pointer )
      a1 ser.rxbuf.len )d d0 .w cmp, eq         ( End of buffer )
      if, 0 #n d0 moveq, then,                  ( Clear the pointer )
   again, ;c                                    ( try next character )

   0 #n d0 moveq,                       ( assume no auto-repeat yet )
   i' system.status a0 move,
   a0 auto )d d1 move,                  ( see if we should be repeating )
   ne if, a0 kticks )d .w tst,
      eq if,  -1 #n d0 moveq,  then,    ( if so, do a repeat )
   then,  d0 sp -) move,  next;         ( and return our flag )

code set-auto  ( -- | turns on auto-repeating of the last key returned )
   i' time0 d0 move,  i' system.status a0 move, d0 a0 kticks )d .w move,
   -1 #n d0 moveq, d0 a0 auto )d move,  next;           ( and turn on flag )

code clear-auto   ( -- | turn off auto-repeating on the last key returned )
   i' system.status a0 move, a0 auto )d clr,  next;

code clear-shifts    ( clears out the shift states )
i' system.status a1 move,               ( Get system area base address )
F4FDFFFD #n a1 keyscan )d andi,
FDFFFFFF #n a1 keyscan 4 + )d andi,
a1 shiftstate )d clr, next;

code do-event  ( -- | process the next event from the keyboard buffer )
   0 #n d1 moveq,               ( clear out d1, so can set lower byte )
   0 #n d3 moveq,               ( clear out d3, so can set lower byte )
 i' system.status a1 move,      ( Get system area base address )
   a1 outptr )d d2 .w move,     ( get the point to read from )
   a1 kb.buf )d a0 .l move,     ( Get the keyboard buffer base address )
   a0 d2 0 xw)d d1 .b move,     ( get the next event from the keyboard )
   1 #n d2 .w addq,             ( Increment the outptr )
   a1 kb.buf.len )d d2 .w cmp, eq       ( End of buffer )
   if, 0 #n d2 moveq,                   ( Clear pointer )
   then, d2 a1 outptr )d .w move,       ( Save kb pointer )
   d1 d2 move,                          ( save event in d2 )
   3f #n d2 .l and,             ( mask off up/down information )
   d2 a1 scancode )d move,      ( save the scancode information )
   i' kcodes a0 move,           ( get the keyboard translation table )
   a1 shiftstate )d d0 move,    ( and the bit vector of shift states )
   d0 d3 move,                  ( we will work with d3 temporarily )
   0e #n d3 andi,               ( mask out all but shift keys & caps lock )
   ne if, 40 #n d2 .b ori, then,        ( if shift or caps lock, use upper case 
table)
   0 #n d0 btst,                ( test for kb1/2 being down )
   ne if, 80 #n d2 .b ori, then,        ( use the second keyboard table )
   d2 d2 .w add,                ( multiply by 2 for wordsized entries )
   a0 d2 0 xl)d d3 .w move,     ( get the translated value )
   d3 d2 .w move,               ( get a copy of the value )
   ff00 #n d2 .w and,           ( keep just the upper nybble )
   ff00 #n d2 .w cmp,           ( and see if it is a shift key )
   eq lif,                      ( if it is a shift key changing )
      d3 d2 .w move,            ( get working copy of translated value )
      07f #n d2 .w andi,        ( and get just the number )
      40 #n d1 .b and,          ( test if shift key was going up or down )
      ne if, d4 sp -) move,     ( save d4 )
         d0 d4 move,
         30 #n d4 andi,         ( see if either uf key is down )
         ne if, 3 #n d2 cmp,    ( if so, see if key going down is caps lock )
            ne if, d2 d0 bset, then,    ( if not, just set the bit )
         else, d2 d0 bset,      ( and set or clear the bit as appropriate )
            1 #n d2 cmp,        ( if the left shift key )
            eq if, 3 #n d0 bclr,        ( if it is, clear the caps lock )
            else, 2 #n d2 cmp,  ( else, see if it is the right shift key )
               eq if, 3 #n d0 bclr, then,       ( also clear the caps lock )
            then,
         then,
         sp )+ d4 move,         ( recover d4 )
      else, 3 #n d2 cmp,        ( see if the key going up is the caps lock )
         ne if, d2 d0 bclr, then,       ( if not, clear the shift key )
      then,
      d0 a1 shiftstate )d move, ( save the new shift state info )
      d0 swap,                  ( and move the info to the upper word )
      d1 d1 .b add,             ( shift the up/down info up to 80 bit )
      80 #n d1 .b eori,         ( and reverse the sense.  80=upgoing now )
      07f #n d3 and,            ( mask the translation down to just the key )
      d1 d3 or,                 ( or in the up/down info to the translation )
      d0 d3 or,                 ( or in the shift state information )
      ff00 #n d3 or,            ( or in the shift key indicator )
      d3 a1 kval )d move,       ( put info into kval )
      -1 #n d3 move,            ( get a true value )
      d3 a1 kstat )d move,      ( and set kstat as having a key ready )
      a1 auto )d clr,           ( and not repeatable )
      d0 swap,                  ( get d0 back to being "shiftstate" )
      3 #n d0 btst,             ( check state of shift lock )
      eq if, fd #n a1 ga3pr.cont )d .b andi,    ( turn off LED )
            a1 ga3pr.cont )d ga3 pr.cont + .b move,
      else, 02 #n a1 ga3pr.cont )d .b ori,      ( turn on LED )
            a1 ga3pr.cont )d ga3 pr.cont + .b move,
      then,
   else,                        ( well, not a shift key )
      40 #n d1 .b andi,         ( find the up/down state )
      eq if,                    ( if going up )
         a1 kval 2 + )d d3 .w cmp,      ( see if same key in lower word )
         eq if, a1 auto )d clr, then,   ( if same, then stop repeating )
      else,                     ( key going down )
         d0 swap,               ( put the shift state in the upper word )
         d0 d3 or,              ( and put with the key information )
         d3 a1 kval )d move,    ( store away the key )
         -1 #n d3 moveq,        ( and get a true value )
         d3 a1 kstat )d move,   ( and set kstat as having a key ready )
      then,
   then, next;                  ( tha tha that's all folk's! )
: @k     ( -- char  \ return the next 'physical' character )
   kval  kstat
   if dup ff00 and ff00 <> auto to
      time0
   else time1 then
   kticks!
   kstat off  ;

: down?  ( n -- flag | true if nth modifier key is down )
   1 swap shl modifiers and 0= 0=  ;

: ?kb2   ( -- flag | true if keyboard II is active, clear if kbd I is active )
   0 down? ;

: ?shift ( -- flag | true if a real shift key is down )
   1 down? 2 down? or ;

: ?shiftlock   ( -- flag | true if shift lock is active )
   3 down? ;

: ?shifted  ( -- flag | true if shift or shift lock is down )
   1 down? 2 down? or 3 down? or ;

: ?ctl      ( -- flag | true if a use-front key is down )
   4 down?  5 down? or ;

: ?lex   ( -- flag | true if the left leap key is down )
   6 down? ;

: ?rex   ( -- flag | true if the right leap key is down )
   7 down? ;

: <<?k>> ( -- flag | true if a real key event available )
   begin kstat if  -1 exit then ( if a real key, then done )
       ?ev while                ( continue while still events to proccess )
      do-event again            ( process the next event )
   playback? if 0 else ?auto then ;     ( if no more, see if should auto-repeat)

: mask   ( n -- n xx | returns mask with nth bit set and all others clear )
   dup 0f and 1 swap shl ;

: !char  ( char -- \ set up to return this as next key )
   dup ff00 and ff00 =                  ( a modifier key? )
   if dup 80 and 0=                     ( determine if up or down modifier key )
      if ?ctl                           ( if a use-front key is down )
         if dup 0f and 3 <>             ( and key is not capslock )
            if mask modifiers or        ( just set key )
            else modifiers then         ( otherwise, don't change state )
         else mask dup modifiers or     ( set the key as going down )
            swap 6 and                  ( see if it was one of the shift keys )
            if 8 not and then           ( if so, clear caps lock )
         then
      else mask dup 8 =                 ( see if the caps is going up )
         if drop modifiers      ( if so, don't modify shift state [ignore it] )
         else not modifiers and then    ( clear the modifier key as going up )
      then modifiers to ( and set back into modifiers )
      modifiers 8 and           ( see if caps lock is on )
      if 2 ga3pr.cont@ or
         ga3pr.cont!
      else fd ga3pr.cont@ and
         ga3pr.cont!
      then
   then
   char to  char? on ;          ( save the char and say we have one )

: sync-shiftkeys  ( --  \ restore shifts to physical state )
   shiftstate modifiers to
   modifiers 8 and toshiftlock ;        ( reset shiftlock LED )

   learnbuff learning? not and
   if thislearn swap drop learnpos > dup 0=
      if clearlearn then
   else 0 then   ;

: toshiftlock  ( flag -- | sets shiftlock to the value of the flag )
   modifiers swap       ( get the flag on the top of the stack )
   if 8 or                      ( set or clear the bit as appropriate )
      2 ga3pr.cont@ or
      ga3pr.cont! ( and set LED )
   else 8 not and
      fd ga3pr.cont@ and
      ga3pr.cont! 
   then
   modifiers to                         ( and save back into the array )
   learnbuff 0=  learning? or           ( if we are not playing back *** )
   if modifiers shiftstate to then ;    ( restore shiftstate *** )

: playback  ( -- char  \ get next playback character )
   thislearn drop learnpos + @  ( get the key back )
   dup 18 shr scancode to               ( take and store the scancode )
   00ffffff and                         ( mask off scancode, this is value )
   dup 10 shr modifiers to              ( also get modifiers off and store )
   4 learnpos +to   ;

: record ( char -- char  \ record a char in learn string )
   learning? learnbuff and
   if thislearn nip learnpos 4 + < not  ( use actual size, instead of maxlearn )
      if dup scancode 18 shl or         ( place scancode in recorded info )
         thislearn drop learnpos + !
         4 learnpos +to
      else clearlearn learnerror on then
   then ;

: <?k>   ( -- f  \ is a character available? )
   char? if -1 exit then        ( if there is a character waiting, done )
   <<?k>>                       ( is there a key ready? )
   if @k                        ( get a physical key event )
      playback?                 ( are we playing back a learn? )
      if dup ff80 and ff80 =    ( is it a shift key going up? )
         if drop char? off      ( if so, just ignore it, & never return it )
         else !char char? off clearlearn then   ( else stop the play back)
      else !char then           ( return the physical key )
   else playback?       ( is there a key to play back? )
      if playback !char von then        ( if so, prepare it )
   then char? ;                 ( did we get a key? )

   char? off                    ( a say we got it )
